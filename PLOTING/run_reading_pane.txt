        def _enable_custom_frame(self):
            self.rootuserinfo.overrideredirect(True)
            self._apply_taskbar_style()

        def _apply_taskbar_style(self):
            root = self.rootuserinfo
            try:
                import ctypes
                user32 = ctypes.windll.user32
                hwnd = user32.GetParent(root.winfo_id())
                style = user32.GetWindowLongW(hwnd, -20)
                user32.SetWindowLongW(hwnd, -20, (style & ~0x00000080) | 0x00040000)
            except Exception:
                return
            root.withdraw()
            root.after(10, lambda: (root.deiconify(), root.lift(), root.focus_force()))

        def _on_map(self, event):
            root = self.rootuserinfo
            if event.widget is root and self.minimizing and root.state() == "normal":
                self.minimizing = False
                root.overrideredirect(True)
                root.after(10, self._apply_taskbar_style)

        def _minimize(self):
            if self.custom_frame:
                self.minimizing = True
                self.rootuserinfo.overrideredirect(False)
            self.rootuserinfo.iconify()

        def toggle_maximize(self):
            root = self.rootuserinfo
            if self.maximized:
                root.geometry(self.normal_geometry)
            else:
                self.normal_geometry = root.geometry()
                x, y, width, height = 0, 0, root.winfo_screenwidth(), root.winfo_screenheight()
                if RP_IS_WINDOWS:
                    try:
                        import ctypes
                        from ctypes import wintypes
                        rect = wintypes.RECT()
                        ctypes.windll.user32.SystemParametersInfoW(0x30, 0, ctypes.byref(rect), 0)
                        x, y, width, height = rect.left, rect.top, rect.right - rect.left, rect.bottom - rect.top
                    except Exception:
                        pass
                root.geometry("%dx%d+%d+%d" % (width, height, x, y))
            self.maximized = not self.maximized
            if hasattr(self, "btn_max"):
                self.btn_max.config(text=self.glyphs["restore" if self.maximized else "max"])

        def _caption_press(self, event):
            self._drag = (event.x_root - self.rootuserinfo.winfo_x(), event.y_root - self.rootuserinfo.winfo_y())

        def _caption_drag(self, event):
            if self.custom_frame and not self.maximized:
                self.rootuserinfo.geometry("+%d+%d" % (event.x_root - self._drag[0], event.y_root - self._drag[1]))

        def _grip_press(self, event):
            root = self.rootuserinfo
            self._resize = (event.x_root, event.y_root, root.winfo_width(), root.winfo_height())

        def _grip_drag(self, event):
            if not self.maximized:
                x0, y0, width, height = self._resize
                self.rootuserinfo.geometry("%dx%d" % (max(760, width + event.x_root - x0),
                                                      max(480, height + event.y_root - y0)))

        def _focus_on_click(self, event):
            """Windows: a window without the Windows title bar does not take the keyboard back by itself once
            another window had it. A click in a text field gives it the keyboard again."""
            if isinstance(event.widget, (tk.Entry, tk.Text)):
                try:
                    event.widget.focus_force()
                except tk.TclError:
                    pass

        def message(self, title, text, icon="info"):
            root = self.rootuserinfo
            win = tk.Toplevel(root)
            win.overrideredirect(True)
            win.configure(bg=RP_FACE, bd=2, relief=tk.RAISED)
            win.transient(root)
            win.attributes("-topmost", True)
            bar = tk.Frame(win, bg=RP_NAVY, height=20)
            bar.pack(fill=tk.X, padx=1, pady=(1, 0))
            bar.pack_propagate(False)
            tk.Label(bar, text=title, bg=RP_NAVY, fg=RP_LIGHT, font=RP_FONT_B).pack(side=tk.LEFT, padx=4)
            self.caption_button(bar, "close", win.destroy).pack(side=tk.RIGHT, padx=2)
            body = tk.Frame(win, bg=RP_FACE)
            body.pack(fill=tk.BOTH, expand=True, padx=14, pady=(12, 6))
            picture = tk.Canvas(body, width=32, height=32, bg=RP_FACE, highlightthickness=0)
            if icon == "error":
                picture.create_oval(2, 2, 30, 30, fill="#ff0000", outline="#800000")
                picture.create_line(10, 10, 22, 22, fill="white", width=3)
                picture.create_line(22, 10, 10, 22, fill="white", width=3)
            else:
                picture.create_oval(2, 2, 30, 30, fill=RP_LIGHT, outline=RP_BLACK)
                picture.create_text(16, 16, text="i", fill="#0000ff", font=("Times New Roman", 16, "bold"))
            picture.pack(side=tk.LEFT, anchor="n", padx=(0, 12))
            tk.Label(body, text=text, bg=RP_FACE, font=RP_FONT, justify="left", wraplength=380).pack(side=tk.LEFT)
            self._button(win, "OK", win.destroy, width=9, default=True).pack(pady=(4, 12))
            win.bind("<Return>", lambda e: win.destroy())
            win.bind("<Escape>", lambda e: win.destroy())
            win.update_idletasks()
            w, h = win.winfo_reqwidth(), win.winfo_reqheight()
            win.geometry("+%d+%d" % (root.winfo_rootx() + max(0, (root.winfo_width() - w) // 2),
                                     root.winfo_rooty() + max(0, (root.winfo_height() - h) // 3)))
            win.lift()
            win.focus_force()

            def take_clicks():
                try:
                    if win.winfo_exists() and win.winfo_viewable():
                        win.grab_set()
                except tk.TclError:
                    pass
            clicks_job = win.after(100, take_clicks)
            # when the box closes (even within 100 ms, or with the whole window) nothing is left waiting
            win.bind("<Destroy>", lambda e, w=win, j=clicks_job: w.after_cancel(j) if e.widget is w else None,
                     add="+")

        def _export_enabled(self):
            """The Export to PDF extension (0x00097), as before."""
            try:
                conn = sqlite3.connect(os.path.join("Config", "dbo_bitpack.db"))
                row = conn.execute("SELECT enabled FROM extensions WHERE extension_id='0x00097'").fetchone()

                return False

        # ─── showing a topic ─────────────────────────────────────────
        def _render(self, text):
            """The guide's plain text, laid out like a help topic (headings, bullets, accounts)."""
            t = self.text
            t.config(state=tk.NORMAL)
            t.delete("1.0", tk.END)
            lines = text.split("\n")
            for index, line in enumerate(lines):
                stripped = line.strip()
                indent = len(line) - len(line.lstrip(" "))
                end = "\n" if index < len(lines) - 1 else ""
                letters = re.sub(r"[^A-Za-z]", "", stripped.split("(")[0])
                if index == 0:
                    t.insert(tk.END, stripped + end, "h1")
                elif stripped and letters and letters.isupper() and not stripped.startswith(("-", "*")):
                    t.insert(tk.END, stripped + end, "h2")
                elif re.match(r"^\d+\.\s", stripped) and indent == 0:
                    t.insert(tk.END, stripped + end, "step")
                elif stripped.startswith(("- ", "* ")):
                    tag = "bullet" if indent == 0 else ("sub" if indent <= 3 else "sub2")
                    symbol = "\u2022 " if stripped.startswith("- ") else "\u00b7 "
                    body = stripped[2:]
                    account = re.match(r"^(\S+/\S*)( - .*)$", body)
                    t.insert(tk.END, symbol, tag)
                    if account:
                        t.insert(tk.END, account.group(1), (tag, "account"))
                        t.insert(tk.END, account.group(2) + end, tag)
                    else:
                        t.insert(tk.END, body + end, tag)
                elif indent > 0 and stripped:
                    t.insert(tk.END, stripped + end, "sub")
                else:
                    t.insert(tk.END, stripped + end)
            t.yview_moveto(0)
            self._highlight_find()

        def open_topic(self, key):
            if key == "features":
                self.show_features()
                return
            short, title, text = TOPIC[key]
            self.current = key
            self.tabrow.pack_forget()
            self.title_band.config(text=title)
            self._render(text)
            self._mark_nav(key)
            self.status.config(text=title)
            self.status_right.config(text="%d lines" % text.count("\n"))

        def _mark_nav(self, key):
            for name, row in self.nav_rows.items():
                selected = name == key or (key in FEATURE_KEYS and name == "features")
                exact = name == key
                row.config(bg=RP_NAVY if exact else RP_LIGHT, fg=RP_LIGHT if exact else RP_BLACK)
                if name == "features":
                    row.config(image=self.icon_book_open if selected else self.icon_book)

        def show_features(self, key=None):
            """The seven feature groups on Windows 95 tabs (the ttk Notebook is gone)."""
            key = key or self.feature_tab

            short, title, text = TOPIC[key]
            self.current = key
            self.title_band.config(text="Features - " + title)
            self.tabrow.pack(fill=tk.X, pady=(4, 0), before=self.text_holder)

                label.config(font=RP_FONT_B if selected else RP_FONT, pady=4 if selected else 2)
                label.pack_configure(pady=(0 if selected else 3, 0))
            self._render(text)
            self._mark_nav(key)
            self.status.config(text="Features - %s" % short)
            self.status_right.config(text="%d lines" % text.count("\n"))

        # ─── Find (in every topic) ───────────────────────────────────
        def _highlight_find(self):
            t = self.text
            t.tag_remove("found", "1.0", tk.END)
            t.tag_remove("found_now", "1.0", tk.END)
            term = self.find_var.get().strip() if hasattr(self, "find_var") else ""
            if not term:
                return 0
            count, start = 0, "1.0"
            while True:
                start = t.search(term, start, stopindex=tk.END, nocase=True)
                if not start:
                    break
                end = "%s+%dc" % (start, len(term))
                t.tag_add("found", start, end)
                start = end
                count += 1
            self.status_right.config(text="%d match(es) here" % count)
            return count

        def find_next(self):
            term = self.find_var.get().strip()
            if not term:
                self.find_entry.focus_force()
                return
            t = self.text
            current = t.tag_ranges("found_now")
            start = current[1] if current else "1.0"
            t.tag_remove("found_now", "1.0", tk.END)
            hit = t.search(term, start, stopindex=tk.END, nocase=True)
            if not hit:
                # not further down this page: the next topic that contains it
                order = [k for k, *_ in TOPICS]
                position = order.index(self.current) if self.current in order else -1
                for step in range(1, len(order) + 1):
                    key = order[(position + step) % len(order)]
                    if term.lower() in TOPIC[key][2].lower():
                        if key in FEATURE_KEYS:
                            self.show_features(key)
                        else:
                            self.open_topic(key)
                        hit = t.search(term, "1.0", stopindex=tk.END, nocase=True)
                        break
            if not hit:
                self.status.config(text="'%s' was not found in the guide." % term)
                return
            end = "%s+%dc" % (hit, len(term))
            t.tag_add("found_now", hit, end)
            t.see(hit)
            self.status.config(text="Found '%s' in %s" % (term, TOPIC[self.current][1]))

        # ─── saving ──────────────────────────────────────────────────
        def save_page_as_text(self):
            """Replaces 'Print Info' (it only showed 'would be implemented here')."""
            short, title, text = TOPIC[self.current]
            path = filedialog.asksaveasfilename(parent=self.rootuserinfo, title="Save Page As",
                                                defaultextension=".txt", initialfile="Bitpack - %s.txt" % short.replace(
                                                    "&", "and").replace("/", "-"),
                                                filetypes=[("Text files", "*.txt"), ("All files", "*.*")])
            if not path:
                return
            try:
                with open(path, "w", encoding="utf-8-sig", newline="\r\n") as fh:
                    fh.write(text + "\n")
                self.status.config(text="Saved: %s" % path)
            except OSError as exc:
                self.message("Save Page As", "The page could not be saved:\n%s" % exc, "error")

        def save_to_pdf(self):
            """All the guide in one PDF, with the name in the top corner (as before); indents are kept now."""
            try:
                from reportlab.lib.pagesizes import letter
                from reportlab.platypus import SimpleDocTemplate, Paragraph, Spacer, PageBreak
                from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle
                from reportlab.lib.enums import TA_LEFT
                from xml.sax.saxutils import escape
            except ImportError:
                self.message("Export to PDF", "The reportlab library is required!\nInstall it with: pip install reportlab",
                             "error")
                return
            filename = filedialog.asksaveasfilename(parent=self.rootuserinfo, defaultextension=".pdf",
                                                    filetypes=[("PDF files", "*.pdf"), ("All files", "*.*")],
                                                    initialfile="Bitpack_UserGuide.pdf")
            if not filename:
                return

            def add_watermark(canvas_obj, doc):
                canvas_obj.saveState()
                canvas_obj.setFont("Helvetica-Bold", 9)
                canvas_obj.setFillColorRGB(0.6, 0.6, 0.6, alpha=0.5)
                canvas_obj.drawRightString(540, 760, "Tudor Marmureanu")
                canvas_obj.drawRightString(580, 750, "@ Retro Computing Division")
                canvas_obj.restoreState()

            try:
                styles = getSampleStyleSheet()
                normal = ParagraphStyle("CustomNormal", parent=styles["Normal"], fontSize=10, leading=14,
                                        alignment=TA_LEFT)
                story = []
                for _key, _short, title, text in TOPICS:
                    story += [Paragraph(escape(title), styles["Heading1"]), Spacer(1, 12)]
                    for line in text.split("\n"):
                        if line.strip():
                            indent = len(line) - len(line.lstrip(" "))
                            style = ParagraphStyle("i%d" % indent, parent=normal, leftIndent=indent * 5)
                            story += [Paragraph(escape(line.strip()), style), Spacer(1, 6)]
                    story.append(PageBreak())
                SimpleDocTemplate(filename, pagesize=letter).build(story, onFirstPage=add_watermark,
                                                                   onLaterPages=add_watermark)
                self.message("Export to PDF", "PDF saved successfully to:\n%s" % filename)
            except Exception as exc:
                self.message("Export to PDF", "Failed to save the PDF:\n%s" % exc, "error")
